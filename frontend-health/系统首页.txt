<script>
  var ecgSlides = Array.prototype.slice.call(document.querySelectorAll('.ecg-slide'));
  var ecgDots   = Array.prototype.slice.call(document.querySelectorAll('.ecg-dot'));
  var ecgTitle  = document.getElementById('ecg-caption-title');
  var ecgDesc   = document.getElementById('ecg-caption-desc');
  var ecgIdx    = 0;

  function updateEcg(index) {
    ecgIdx = (index + ecgSlides.length) % ecgSlides.length;
    ecgSlides.forEach(function(s, i) { s.classList.toggle('active', i === ecgIdx); });
    ecgDots.forEach(function(d, i) { d.classList.toggle('active', i === ecgIdx); });
    var active = ecgSlides[ecgIdx];
    ecgTitle.textContent = active.getAttribute('data-title');
    ecgDesc.textContent  = active.getAttribute('data-description');
  }

  document.getElementById('ecg-prev').addEventListener('click', function() { updateEcg(ecgIdx - 1); });
  document.getElementById('ecg-next').addEventListener('click', function() { updateEcg(ecgIdx + 1); });
  ecgDots.forEach(function(d, i) { d.addEventListener('click', function() { updateEcg(i); }); });

  setInterval(function() { updateEcg(ecgIdx + 1); }, 5000);
  updateEcg(0);
</script>
